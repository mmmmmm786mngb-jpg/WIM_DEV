// q01: Документы ПланРегламентныхОперацийДУ за сутки по пулу или по списку портфелей
//
// Параметры (консоль запросов):
//   &ДатаНач            - Дата (начало дня операционного дня)
//   &ДатаКон            - Дата (конец дня операционного дня)
//   &ТолькоПул          - Булево: Истина = все портфели указанного &Пул; Ложь = режим списка
//   &Пул                - СправочникСсылка.Пул (обязателен при &ТолькоПул = Истина)
//   &ТЗПортфелей        - ТаблицаЗначений с колонкой "Портфель" (тип СправочникСсылка.Портфели)
//                         при &ТолькоПул = Ложь; пустые строки не передавать или отфильтровать
//   &ФильтрПулВСписке   - Булево: при &ТолькоПул = Ложь и Истина — оставить только портфели с Пул = &Пул
//
// Отбор: портфели и планы без пометки удаления (НЕ ПометкаУдаления).
//
// Примеры:
//   Режим пула:  ТолькоПул = Истина, задать Пул, ТЗПортфелей — пустая таблица с колонкой Портфель
//   Режим списка: ТолькоПул = Ложь, заполнить ТЗПортфелей; Пул и ФильтрПулВСписке — по необходимости

////////////////////////////////////////////////////////////////////////////////
// Временная таблица портфелей отбора
////////////////////////////////////////////////////////////////////////////////

ВЫБРАТЬ
	П.Ссылка КАК Портфель
ПОМЕСТИТЬ ВТ_ПортфелиОтбора
ИЗ
	Справочник.Портфели КАК П
ГДЕ
	&ТолькоПул
	И НЕ П.ПометкаУдаления
	И П.Пул = &Пул

ОБЪЕДИНИТЬ ВСЕ

ВЫБРАТЬ
	Т.Портфель КАК Портфель
ИЗ
	&ТЗПортфелей КАК Т
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник.Портфели КАК П
		ПО Т.Портфель = П.Ссылка
ГДЕ
	НЕ &ТолькоПул
	И НЕ П.ПометкаУдаления
	И ЗначениеЗаполнено(Т.Портфель)
	И (НЕ &ФильтрПулВСписке ИЛИ П.Пул = &Пул)
;

////////////////////////////////////////////////////////////////////////////////
// Планы за период по отбору портфелей
////////////////////////////////////////////////////////////////////////////////

ВЫБРАТЬ
	План.Ссылка КАК ПланСсылка,
	План.Дата КАК Дата,
	План.Номер КАК Номер,
	План.Портфель КАК Портфель,
	План.Проведен КАК Проведен,
	План.ГруппаОперацийПользователя КАК ГруппаОперацийПользователя
ИЗ
	Документ.ПланРегламентныхОперацийДУ КАК План
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ ВТ_ПортфелиОтбора КАК Отбор
		ПО План.Портфель = Отбор.Портфель
ГДЕ
	План.Дата МЕЖДУ &ДатаНач И &ДатаКон
	И НЕ План.ПометкаУдаления

УПОРЯДОЧИТЬ ПО
	План.Портфель,
	План.Дата,
	План.Номер
