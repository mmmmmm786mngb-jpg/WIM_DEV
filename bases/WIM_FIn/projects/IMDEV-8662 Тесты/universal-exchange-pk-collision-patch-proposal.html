Патч: дубль PRIMARY KEY при параллельной загрузке

Контекст: два потока создают запись с одним GUID, первый успевает записать объект, второй получает ошибку SQL по PRIMARY KEY. Ниже приведен итоговый вариант для WIM_FIn, который проверен на сценарии БизнесПроцессСсылка.СделкиTn.

1. Где править

WIM_FIn: C:\1c\Cursor_1c\WORK\WIM_FIn\src\cf\DataProcessors\УниверсальныйОбменДаннымиXML\Ext\ObjectModule.bsl
Процедура ЗаписатьОбъектВИБ.

Wim_Du: аналогичный блок, но текст ошибки брать через ОписаниеОшибки().

			Объект, Тип);
		
		Если Не ФлагРежимОтладки Тогда
			ВызватьИсключение СтрокаСообщенияОбОшибке;
		КонецЕсли;
		
	КонецПопытки;

3. Матрица решения после патча

СостояниеДействие
Объект неполный Ошибку не считаем критичной, выходим без исключения
Объект полный, ОбъектВБ неполный Удаляем ОбъектВБ, повторно записываем Объект, выходим без исключения
Объект полный, ОбъектВБ полный Оставляем стандартную ошибку как в исходнике
Для поиска объекта в базе используется GUID из текста SQL-ошибки: извлекается 0x..., переводится в строку GUID 1С, затем берется ссылка БизнесПроцессы.СделкиTn.ПолучитьСсылку(...).

4. Процедура ЗаписатьОбъектВИБ(Объект, Тип) после патча

Процедура ЗаписатьОбъектВИБ(Объект, Тип)
	
	Попытка
		
		УстановитьОбменДаннымиЗагрузка(Объект);
		Объект.Записать();
		
	Исключение
		
		ИнфоОшибки = ИнформацияОбОшибке();
		ТекстОшибки = ОбработкаОшибок.ПодробноеПредставлениеОшибки(ИнфоОшибки);
		Верх = ВРег(ТекстОшибки);
		
		ЭтоОшибкаДубля =
			СтрНайти(Верх, "PRIMARY KEY") > 0
			Или СтрНайти(Верх, "UNIQUE KEY") > 0
			Или СтрНайти(Верх, "UNIQUE CONSTRAINT") > 0
			Или СтрНайти(Верх, "НАРУШЕНИЕ ОГРАНИЧЕНИЯ UNIQUE") > 0
			Или СтрНайти(Верх, "НАРУШЕНИЕ ОГРАНИЧЕНИЯ PRIMARY KEY") > 0
			Или СтрНайти(Верх, "UNIQUE VIOLATION") > 0;
		
		ОбработалиДубль = Ложь;
		ТекстОшибкиПатчинга = "";
		
		Если ЭтоОшибкаДубля Тогда
			ПозицияНач = СтрНайти(Верх, "0X");
			Если ПозицияНач > 0 Тогда
				Hex = Сред(Верх, ПозицияНач + 2, 32);
				Hex = СтрЗаменить(Hex, " ", "");
				Если СтрДлина(Hex) = 32 Тогда
					GUIDСтрокой =
						НРег(Сред(Hex, 25, 8)) + "-" +
						НРег(Сред(Hex, 21, 4)) + "-" +
						НРег(Сред(Hex, 17, 4)) + "-" +
						НРег(Сред(Hex, 1, 4)) + "-" +
						НРег(Сред(Hex, 5, 12));
					
					Ссылка = БизнесПроцессы.СделкиTn.ПолучитьСсылку(Новый УникальныйИдентификатор(GUIDСтрокой));
					ОбъектВБ = Ссылка.ПолучитьОбъект();
					
					Если Не ОбъектВБ.ЭтоНовый() Тогда
						ОбъектПолный = ЗначениеЗаполнено(Объект.Актив) И ЗначениеЗаполнено(Объект.Количество);
						ОбъектВБПолный = ЗначениеЗаполнено(ОбъектВБ.Актив) И ЗначениеЗаполнено(ОбъектВБ.Количество);
						
						Если Не ОбъектПолный Тогда
							ОбработалиДубль = Истина;
						ИначеЕсли Не ОбъектВБПолный Тогда
							Попытка
								УстановитьОбменДаннымиЗагрузка(ОбъектВБ);
								ОбъектВБ.Удалить();
								УстановитьОбменДаннымиЗагрузка(Объект);
								Объект.Записать();
								ОбработалиДубль = Истина;
							Исключение
								ТекстОшибкиПатчинга = ОбработкаОшибок.ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
								ОбработалиДубль = Ложь;
							КонецПопытки;
						КонецЕсли;
					КонецЕсли;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
		
		Если ОбработалиДубль Тогда
			Возврат;
		КонецЕсли;
		Если ЗначениеЗаполнено(ТекстОшибкиПатчинга) Тогда
			ТекстОшибки = ТекстОшибки + Символы.ПС + "После попытки патчинга ошибка такая:" + Символы.ПС + ТекстОшибкиПатчинга;
		КонецЕсли;
		
		СтрокаСообщенияОбОшибке = ЗаписатьИнформациюОбОшибкеВПротокол(26, ТекстОшибки, Объект, Тип);
		
		Если Не ФлагРежимОтладки Тогда
			ВызватьИсключение СтрокаСообщенияОбОшибке;
		КонецЕсли;
		
	КонецПопытки;
	
КонецПроцедуры
