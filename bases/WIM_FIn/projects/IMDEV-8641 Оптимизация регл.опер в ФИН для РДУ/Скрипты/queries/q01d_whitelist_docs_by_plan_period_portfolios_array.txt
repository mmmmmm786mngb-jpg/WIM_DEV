// q01d_array: То же, что q01d_whitelist_docs_by_plan_period_portfolios.txt, но список портфелей задаётся
//             параметром-массивом &МассивПортфелей (для консоли запросов), а не таблицей значений.
//
// Белый список документов — как в СоответствиеПроцедурИИменДокументов (ObjectModule.bsl EPF).
// Связь: ДокументОснование = план; см. предупреждение про КритерииОтбора.СвязанныеДокументы в q01d.
//
// Параметры:
//   &ДатаНач, &ДатаКон     - границы периода по дате документа плана
//   &ТолькоПул             - Булево: Истина = все непомеченные портфели пула &Пул; Ложь = режим массива
//   &Пул                   - СправочникСсылка.Пул (для режима пула; для фильтра в режиме массива при &ФильтрПулВСписке)
//   &МассивПортфелей       - Массив из СправочникСсылка.Портфели (при &ТолькоПул = Ложь). При &ТолькоПул = Истина
//                            допустим пустой массив — вторая ветка ОБЪЕДИНИТЬ не даст строк
//   &ФильтрПулВСписке      - Булево: при &ТолькоПул = Ложь и Истина — из массива остаются только портфели с Пул = &Пул
//
// Отбор: портфели и планы без пометки удаления; подчинённые документы белого списка — без пометки удаления.
//
// Пример установки параметра в коде перед выполнением запроса:
//   МассивПортфелей = Новый Массив;
//   МассивПортфелей.Добавить(Справочники.Портфели.НайтиПоКоду("..."));
//   Запрос.УстановитьПараметр("МассивПортфелей", МассивПортфелей);

////////////////////////////////////////////////////////////////////////////////
// ВТ портфелей: пул или элементы из массива
////////////////////////////////////////////////////////////////////////////////

ВЫБРАТЬ
	П.Ссылка КАК Портфель
ПОМЕСТИТЬ ВТ_ПортфелиОтбора
ИЗ
	Справочник.Портфели КАК П
ГДЕ
	&ТолькоПул
	И НЕ П.ПометкаУдаления
	И П.Пул = &Пул

ОБЪЕДИНИТЬ ВСЕ

ВЫБРАТЬ
	П.Ссылка КАК Портфель
ИЗ
	Справочник.Портфели КАК П
ГДЕ
	НЕ &ТолькоПул
	И НЕ П.ПометкаУдаления
	И П.Ссылка В (&МассивПортфелей)
	И (НЕ &ФильтрПулВСписке ИЛИ П.Пул = &Пул)
;

ИЗ
	Документ.ПланРегламентныхОперацийДУ КАК План

		ПО План.Портфель = Отбор.Портфель
ГДЕ
	План.Дата МЕЖДУ &ДатаНач И &ДатаКон
	И НЕ План.ПометкаУдаления
;
