4. Функция NDFL_RDU_ОбщиеОбороты(ПараметрыРасчета, ПрошлыйПериод = Ложь)

	// IMDEV-7330 Нормализуем мультипортфельные параметры ДО установки периодов запроса.
	// Это важно, чтобы МинимальнаяДатаДокументов и связанные таблицы были рассчитаны на тот же набор портфелей.
	////////////////NDFL_RDU_ПараметрыРасчета = NDFL_RDU_НормализоватьПараметрыДляОбщихОборотов(ПараметрыРасчета, Портфели);
	
	ПараметрыРасчета.Вставить("Портфели", Портфели);
	NDFL_RDU_ПараметрыРасчета = Новый Структура;
	NDFL_RDU_ПараметрыРасчета.Вставить("РасчетСуммыНДФЛВыполнятьПоСуммамНалоговогоУчета",
		NDFL_RDU_НастройкиРасчетаНУПоПортфелям(Портфели));
	
	ДанныеСпискаДокументов = NDFL_RDU_СписокДокументовДляРасчетаДоходовРасходов(ПараметрыРасчета, Истина); ///////////////////////////////////
	NDFL_RDU_ПараметрыРасчета.Вставить("СписокДокументовДляРасчетаДоходовРасходов", ДанныеСпискаДокументов.СписокДокументов);
	NDFL_RDU_ПараметрыРасчета.Вставить("МинимальнаяДатаДокументов", ДанныеСпискаДокументов.МинимальнаяДатаДокументов);
	
	NDFL_RDU_ПараметрыРасчета.Вставить("ТаблицаПродажДляРасчетаЛьготПоКлиентамПодСанкциями",
		NDFL_RDU_ТаблицаПродажДляРасчетаЛьготПоКлиентамПодСанкциями(ПараметрыРасчета));
	// < --
	
	
	Если ПрошлыйПериод Тогда
		Запрос.УстановитьПараметр("НачалоПериода", НачалоГода(ПараметрыРасчета.НачалоПериода-1));
		Запрос.УстановитьПараметр("КонецПериода", ПараметрыРасчета.НачалоПериода-1);
	Иначе	
		Запрос.УстановитьПараметр("НачалоПериода", NDFL_RDU_ПараметрыРасчета.МинимальнаяДатаДокументов);
		Запрос.УстановитьПараметр("КонецПериода", МоментВремени());
	КонецЕсли;

	// IMDEV-7330 В параметр Портфель передаем Массив (используется в тексте запроса как В (&Портфель)).
	Запрос.УстановитьПараметр("ДатаРасчетаНДФЛ", Дата);
	Запрос.УстановитьПараметр("Портфель", Портфели);
	// IMDEV-7330 Передаем ТаблицаЗначений: (Портфель, РасчетСуммыНДФЛВыполнятьПоСуммамНалоговогоУчета).
	Запрос.УстановитьПараметр("РасчетСуммыНДФЛВыполнятьПоСуммамНалоговогоУчета", NDFL_RDU_ПараметрыРасчета.РасчетСуммыНДФЛВыполнятьПоСуммамНалоговогоУчета);
	// IMDEV-7330 Передаем ТаблицаЗначений: (Портфель, ДокументДляРасчетаДоходовРасходов).
	Запрос.УстановитьПараметр("СписокДокументовДляРасчетаДоходовРасходов", NDFL_RDU_ПараметрыРасчета.СписокДокументовДляРасчетаДоходовРасходов);
	// IMDEV-7330 Передаем ТаблицаЗначений с детализацией продаж по санкционным клиентам.
	Запрос.УстановитьПараметр("ТаблицаПродажДляРасчетаЛьготПоКлиентамПодСанкциями", NDFL_RDU_ПараметрыРасчета.ТаблицаПродажДляРасчетаЛьготПоКлиентамПодСанкциями);
	Запрос.УстановитьПараметр("Выручка", ПланыСчетов.Хозрасчетный.Выручка);
	Запрос.УстановитьПараметр("РасчетыЦБПродажа", ПланыСчетов.Хозрасчетный.РасчетыЦБПродажа);
	Запрос.УстановитьПараметр("АкцииТело", ПланыСчетов.Хозрасчетный.АкцииТело);
	Запрос.УстановитьПараметр("Себестоимость", ПланыСчетов.Хозрасчетный.Себестоимость);
	Запрос.УстановитьПараметр("СчетаУчета", ПланыСчетов.Хозрасчетный.СчетаПоСтроке("68.01, 90.01, 90.02, 91.01.02, 91.01.09, 91.02.02, 91.02.03, 91.02.04, 91.02.05, 91.02.06, 91.02.10, 91.02.11, 91.03, 91.04, 91.02.08"));
	Запрос.УстановитьПараметр("СчетаСрочныеКонтракты",ПланыСчетов.Хозрасчетный.СчетаПоСтроке("76.01.02,55.03.2"));
	ОбщегоНазначенияДУ.ДобавитьВЛогОтладки(Запрос, "НачислениеНДФЛПоПортфелю.ОбщиеОбороты");
	
	ОбщиеОбороты = Запрос.Выполнить().Выгрузить();
	
	Возврат ОбщиеОбороты;
