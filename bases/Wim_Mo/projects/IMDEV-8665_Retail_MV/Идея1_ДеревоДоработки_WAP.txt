// Получает данные для одного конкретного мандата из общей структуры,
// полученной через массовый вызов ПолучитьДанныеРасчета()
// IMDEV-8665-MV
Функция ПолучитьДанныеМандата(СтруктураДанныхРасчета, Мандат) Экспорт
	
	Если СтруктураДанныхРасчета = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;
	
	ДанныеМандата = Новый Структура;
	
	// 1. Данные, которые не зависят от мандата
	ДанныеМандата.Вставить("ВалютыРасчета", СтруктураДанныхРасчета.ВалютыРасчета);
	ДанныеМандата.Вставить("ТаблицаКотировок", СтруктураДанныхРасчета.ТаблицаКотировок);
	
	// 2. ЭтоНачалоОтчетногоПериода (булево)
	Если ТипЗнч(СтруктураДанныхРасчета.ЭтоНачалоОтчетногоПериода) = Тип("Соответствие") Тогда
		Значение = СтруктураДанныхРасчета.ЭтоНачалоОтчетногоПериода.Получить(Мандат);
		ДанныеМандата.Вставить("ЭтоНачалоОтчетногоПериода", ?(Значение = Неопределено, Ложь, Значение));
	Иначе
		ДанныеМандата.Вставить("ЭтоНачалоОтчетногоПериода", Ложь);
	КонецЕсли;
	
	// 3. Таблицы, которые нужно фильтровать по мандату
	ДанныеМандата.Вставить("ТаблицаПредыдущихСтоимостей", 
		ОтфильтроватьТаблицуПоМандату(СтруктураДанныхРасчета.ТаблицаПредыдущихСтоимостей, Мандат));
	
	ДанныеМандата.Вставить("ТаблицаОпераций", 
		ОтфильтроватьТаблицуПоМандату(СтруктураДанныхРасчета.ТаблицаОпераций, Мандат));
	
	ДанныеМандата.Вставить("РезультатЗапроса", 
		ОтфильтроватьТаблицуПоМандату(СтруктураДанныхРасчета.РезультатЗапроса, Мандат));
	
	Возврат ДанныеМандата;

	ЭтоМассив = ТипЗнч(Мандат) = Тип("Массив") Или ТипЗнч(Мандат) = Тип("СписокЗначений");
	
	Запрос = Новый Запрос;

	СтруктураВозврата.Вставить("ТаблицаПредыдущихСтоимостей",РезультатВыполнения[...]);
	СтруктураВозврата.Вставить("ТаблицаОпераций",            РезультатВыполнения[...]);

// Универсальная фильтрация таблицы по мандату
Функция ОтфильтроватьТаблицуПоМандату(Знач Таблица, Знач Мандат) Экспорт
	
	Если Таблица = Неопределено Или Таблица.Количество() = 0 Тогда
		Возврат Новый ТаблицаЗначений;
	КонецЕсли;
	
	// Определяем имя колонки с мандатом
	ИмяКолонкиМандата = "";
	Если Таблица.Колонки.Найти("Мандат") <> Неопределено Тогда
		ИмяКолонкиМандата = "Мандат";
	ИначеЕсли Таблица.Колонки.Найти("СчетМандата") <> Неопределено Тогда
		ИмяКолонкиМандата = "СчетМандата";
	Иначе
		// Если колонки нет — возвращаем копию всей таблицы
		Возврат Таблица.Скопировать();
	КонецЕсли;
	
	Отбор = Новый Структура(ИмяКолонкиМандата, Мандат);
	Возврат Таблица.Скопировать(Отбор);
