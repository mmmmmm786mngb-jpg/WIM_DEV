// Универсальная функция — используется во всех процедурах заполнения
Функция ОпределитьМножествоМандатов() Экспорт
	
	Если ДополнительныеСвойства.Свойство("МассивМандатов")
		И ЗначениеЗаполнено(ДополнительныеСвойства.МассивМандатов) Тогда
		Возврат ДополнительныеСвойства.МассивМандатов;
	КонецЕсли;
	
	Возврат Мандат; // обычный режим (один мандат)
	
КонецФункции

// Главный метод — заполняет все ТЧ по массиву мандатов сразу
Процедура ПерезаполнитьВсеТЧ(ПоВсемМандатам = Ложь) Экспорт
	
	Если ПоВсемМандатам Тогда
		ДополнительныеСвойства.Вставить("ЭтоСлужебныйРежим", Истина);
	Иначе
		ДополнительныеСвойства.Вставить("ЭтоСлужебныйРежим", Ложь);
	КонецЕсли;
	
	// Вызываем все процедуры заполнения
	ЗаполнитьТабличнуюЧасть_Служебная();
	ЗаполнитьТЧЦенныеБумаги();
	ЗаполнитьТЧПлановыеЦБ();
	ЗаполнитьТЧНКД();
	ЗаполнитьТЧПлановыйНКД();
	
	// Если у тебя есть другие ТЧ, которые зависят от мандата — добавь их сюда
	
КонецПроцедуры

// Строит быстрый индекс по мандатам (вызывается один раз после ПерезаполнитьВсеТЧ)
Процедура ПостроитьИндексПоМандатам() Экспорт
	
	Если ДополнительныеСвойства.Свойство("ИндексМандатов") Тогда
		Возврат;
	КонецЕсли;
	
	Индекс = Новый Соответствие;
	
	Для Каждого МетаданныеТЧ Из Метаданные().ТабличныеЧасти Цикл
		
		Если МетаданныеТЧ.Реквизиты.Найти("СчетМандата") = Неопределено Тогда
			Продолжить;
		КонецЕсли;
		
		ИмяТЧ = МетаданныеТЧ.Имя;
		ТЧ = ЭтотОбъект[ИмяТЧ];
		
		Если ТЧ.Количество() = 0 Тогда
			Продолжить;
		КонецЕсли;
		
		ВремИндекс = Новый Соответствие;
		
		Для Каждого Стр Из ТЧ Цикл
			Если ЗначениеЗаполнено(Стр.СчетМандата) Тогда
				Мандат = Стр.СчетМандата.Владелец;
				Если ВремИндекс[Мандат] = Неопределено Тогда
					ВремИндекс[Мандат] = Новый Массив;
				КонецЕсли;
				ВремИндекс[Мандат].Добавить(Стр);
			КонецЕсли;
		КонецЦикла;
		
		// Формируем готовые таблицы значений
		Для Каждого КЗ Из ВремИндекс Цикл
			Ключ = ИмяТЧ + "_" + Строка(КЗ.Ключ.УникальныйИдентификатор());
			Индекс[Ключ] = ТЧ.Выгрузить(КЗ.Значение);
		КонецЦикла;
		
	КонецЦикла;
	
	ДополнительныеСвойства.Вставить("ИндексМандатов", Индекс);
	
КонецПроцедуры

// Копирует данные по одному мандату из служебного документа в обычный
Процедура КопироватьПоМандату(Служебный, Док, Мандат) Экспорт
	
	Индекс = Служебный.ДополнительныеСвойства.Получить("ИндексМандатов");
	Если Индекс = Неопределено Тогда
		Возврат;
	КонецЕсли;
	
	Для Каждого МетаданныеТЧ Из Служебный.Метаданные().ТабличныеЧасти Цикл
		
		Если МетаданныеТЧ.Реквизиты.Найти("СчетМандата") = Неопределено Тогда
			Продолжить;
		КонецЕсли;
		
		ИмяТЧ = МетаданныеТЧ.Имя;
		Ключ  = ИмяТЧ + "_" + Строка(Мандат.УникальныйИдентификатор());
		
		Данные = Индекс.Получить(Ключ);
		
		Док[ИмяТЧ].Очистить();
		Если Данные <> Неопределено Тогда
			Док[ИмяТЧ].Загрузить(Данные);
		КонецЕсли;
		
	КонецЦикла;

Функция ЗаменитьМандатНаВ(Знач ТекстЗапроса) Экспорт
	
	РегВыр = Новый РегулярноеВыражение("\s*=\s*\&Мандат\b");
	РегВыр.ИгнорироватьРегистр = Истина;
	
	Возврат РегВыр.Заменить(ТекстЗапроса, " В (&Мандат)");
	
КонецФункции
