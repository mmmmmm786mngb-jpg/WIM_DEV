// Универсальная параллельная процедура создания и проведения документов
// Параметры:
//   ИмяДокумента - Строка - "РасчетРыночнойСтоимости" или "РасчетСредневзвешеннойСтоимости"
//   ТекДата      - Дата
//   ТЗМандатов   - ТаблицаЗначений (с колонкой Мандат обязательно)
//   Ошибка       - Булево (возвращаемый)
//   МаксПотоков  - Число - максимальное количество параллельных заданий (по умолчанию 10)
Процедура СформироватьДокументыПараллельно(ИмяДокумента, ТекДата, ТЗМандатов, Ошибка, МаксПотоков = 10) Экспорт
	
	Ошибка = Ложь;
	АктивныеЗадания = Новый Массив;
	Индекс = 0;
	Количество = ТЗМандатов.Количество();
	
	МаксПотоков = Макс(МаксПотоков, 1); // защита от некорректного значения
	
	Пока Индекс < Количество Цикл
		
		// Запускаем новые задания
		Пока АктивныеЗадания.Количество() < МаксПотоков И Индекс < Количество Цикл
			
			Строка = ТЗМандатов[Индекс];
			
			ПараметрыЗадания = Новый Структура;
			ПараметрыЗадания.Вставить("ИмяДокумента", ИмяДокумента);
			ПараметрыЗадания.Вставить("ТекДата",      ТекДата);
			ПараметрыЗадания.Вставить("Мандат",       Строка.Мандат);
			ПараметрыЗадания.Вставить("Валюта",       ?(Строка.Свойство("Валюта"), Строка.Валюта, Неопределено));
			ПараметрыЗадания.Вставить("Курс",         ?(Строка.Свойство("Курс"), Строка.Курс, 1));
			ПараметрыЗадания.Вставить("Кратность",    ?(Строка.Свойство("Кратность"), Строка.Кратность, 1));
			
			ПараметрыФЗ = Новый Структура;
			ПараметрыФЗ.Вставить("ИмяМетода", "СоздатьИПровестиДокументВФоне");
			ПараметрыФЗ.Вставить("ПараметрыВыполнения", ПараметрыЗадания);
			ПараметрыФЗ.Вставить("ЭтоВнешняяОбработка", Истина);
			ПараметрыФЗ.Вставить("ДополнительнаяОбработкаСсылка", ПолучитьСсылкуНаСебя());
			ПараметрыФЗ.Вставить("ИмяОбработки", "внРасчетMVиWAP");
			
			Результат = ДлительныеОперации.ЗапуститьВыполнениеВФоне(
				Новый УникальныйИдентификатор,
				"ДлительныеОперации.ВыполнитьПроцедуруМодуляОбъектаОбработки",
				ПараметрыФЗ,
				"Формирование " + ИмяДокумента + " по мандату " + Строка.Мандат);
			
			АктивныеЗадания.Добавить(Результат);
			Индекс = Индекс + 1;
		КонецЦикла;
		
		// Мониторинг с паузой
		Если АктивныеЗадания.Количество() > 0 Тогда
			ОбщегоНазначенияКлиентСервер.Пауза(0.1);
			
			Для i = АктивныеЗадания.ВГраница() По 0 Цикл
				З = АктивныеЗадания[i];
				Если З.Свойство("ИдентификаторЗадания") Тогда
					ФЗ = ФоновыеЗадания.НайтиПоУникальномуИдентификатору(З.ИдентификаторЗадания);
					Если ФЗ = Неопределено Или ФЗ.Состояние <> СостояниеФоновогоЗадания.Активно Тогда
						АктивныеЗадания.Удалить(i);
					КонецЕсли;
				КонецЕсли;
			КонецЦикла;
		КонецЕсли;
		
	КонецЦикла;
	
	// Ожидаем завершения оставшихся заданий
	Если АктивныеЗадания.Количество() > 0 Тогда
		Идентификаторы = Новый Массив;
		Для Каждого З Из АктивныеЗадания Цикл
			Если З.Свойство("ИдентификаторЗадания") Тогда
				Идентификаторы.Добавить(З.ИдентификаторЗадания);
			КонецЕсли;
		КонецЦикла;
		Если Идентификаторы.Количество() > 0 Тогда
			ФоновыеЗадания.ОжидатьЗавершения(Идентификаторы);
		КонецЕсли;
	КонецЕсли;
	
	// Проверка на ошибки
	Для Каждого З Из АктивныеЗадания Цикл
		Если З.Свойство("ИдентификаторЗадания") Тогда
			ФЗ = ФоновыеЗадания.НайтиПоУникальномуИдентификатору(З.ИдентификаторЗадания);
			Если ФЗ <> Неопределено И ФЗ.Состояние = СостояниеФоновогоЗадания.ЗавершеноАварийно Тогда
				Ошибка = Истина;
			КонецЕсли;
		КонецЕсли;
	КонецЦикла;
