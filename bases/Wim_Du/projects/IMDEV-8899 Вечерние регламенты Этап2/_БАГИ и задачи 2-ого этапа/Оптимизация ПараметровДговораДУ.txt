Оптимизация ПараметровДговораДУ
Она может вначале вызвать закешированную функцию которая только по ДоговорДУ и Дате получает параметры



// Функция возвращает параметры ДоговораДУ в разрезе вариантов
//
// Параметры:
//  ДоговорДУ - СправочникСсылка.ДоговорДУ
//  Параметры - Массив, ПланВидовХарактеристикСсылка.ПараметрыДоговораДУ - параметры
//  Дата      - Дата
//
// Возвращаемое значение:
//  Соответствие - Параметр -> Значение (НЕОПРЕДЕЛЕНО, если значения нет)
//
Функция ЗначенияПараметровДоговораДУ(ДоговорДУ, Параметры, Знач Дата = '00010101') Экспорт
    
    ЗначенияПараметров = Новый Соответствие;
    
    Если Параметры = Неопределено Тогда
        Возврат ЗначенияПараметров;
    КонецЕсли;
    
    Если ТипЗнч(Параметры) <> Тип("Массив") Тогда
        МассивПараметров = Новый Массив;
        МассивПараметров.Добавить(Параметры);
    Иначе
        МассивПараметров = Параметры;
    КонецЕсли;
    
    Если Дата = '00010101' Тогда
        Дата = ТекущаяДата();
    КонецЕсли;
    
    // === ОПТИМИЗИРОВАННЫЙ ЗАПРОС ===
    Запрос = Новый Запрос;
    Запрос.УстановитьПараметр("Дата", Дата);
    Запрос.УстановитьПараметр("ДоговорДУ", ДоговорДУ);
    
    Запрос.Текст = 
    "ВЫБРАТЬ
    |   Срез.Параметр КАК Параметр,
    |   Срез.Значение КАК Значение
    |ИЗ
    |   РегистрСведений.ЗначенияПараметровДоговораДУ.СрезПоследних(
    |       &Дата, 
    |       ДоговорДУ = &ДоговорДУ
    |   ) КАК Срез";
    
    // Логирование оставляем только при необходимости (см. ниже)
    // ОбщегоНазначенияДУ.ДобавитьВЛогОтладки(Запрос, "РегистрСведений.ЗначенияПараметровДоговоровДУ.МодульМенеджера:ЗначенияПараметровДоговораДУ");
    
    Результат = Запрос.Выполнить();
    Выборка = Результат.Выбрать();
    
    // Кладём все имеющиеся значения во временное соответствие
    СоответствиеВсехЗначений = Новый Соответствие;
    Пока Выборка.Следующий() Цикл
        СоответствиеВсехЗначений.Вставить(Выборка.Параметр, Выборка.Значение);
    КонецЦикла;
    
    // Формируем результат только по запрошенным параметрам
    Для Каждого Параметр Из МассивПараметров Цикл
        Значение = СоответствиеВсехЗначений.Получить(Параметр);
        // Если параметра нет в регистре — будет Неопределено (это и нужно)
        ЗначенияПараметров.Вставить(Параметр, Значение);
    КонецЦикла;
    
    Возврат ЗначенияПараметров;
